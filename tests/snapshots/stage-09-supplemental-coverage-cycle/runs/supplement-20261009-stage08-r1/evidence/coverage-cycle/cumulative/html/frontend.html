166 function defaultTimes(day) {
167   const now = new Date(), localTime = timeString(now);
168   let minutes = day === dateString(now) ? Math.ceil((Number(localTime.slice(0, 2)) * 60 + Number(localTime.slice(3)) + 1) / 15) * 15 : 9 * 60;
169   let chosenDay = day;
170   if (day < dateString(now) || minutes >= 20 * 60) {
171     chosenDay = dateString(new Date(now.getTime() + 86400000)); minutes = 9 * 60;
172   }
173   minutes = Math.max(8 * 60, minutes);
174   const format = value => `${String(Math.floor(value / 60)).padStart(2, '0')}:${String(value % 60).padStart(2, '0')}`;
175   return { day: chosenDay, start: format(minutes), end: format(Math.min(minutes + 60, 20 * 60)) };
176 }
